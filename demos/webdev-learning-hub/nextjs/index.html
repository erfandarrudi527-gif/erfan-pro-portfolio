import { Suspense } from 'react';
import { orderServerAction } from './actions';
import ServerMetricsSkeleton from './components/Skeleton';

// Direct Database Fetch in Server Component (No useEffect, No Waterfalls)
async function getEnterpriseProducts() {
  // Runs exclusively on the Node/Edge server
  const res = await fetch('https://api.cloudscale.dev/products', {
    next: { tags: ['products'], revalidate: 3600 } // ISR Cache Tagging
  });
  if (!res.ok) throw new Error('Failed to fetch enterprise catalog');
  return res.json();
}

export default async function HomePage() {
  const products = await getEnterpriseProducts();

  return (
    <main className="enterprise-layout">
      <header className="page-header">
        <span className="badge">Next.js 15 App Router</span>
        <h1>زیرساخت ابری مقیاس سازمانی CloudScale</h1>
      </header>

      {/* Instant Partial Streaming via Suspense */}
      <Suspense fallback={<ServerMetricsSkeleton />}>
        <section className="catalog-grid">
          {products.map(prod => (
            <article key={prod.id} className="product-card">
              <h3>{prod.name}</h3>
              <p className="price">{prod.priceFormatted}</p>
              
              {/* Native Server Action Form with Zero Client State */}
              <form action={orderServerAction}>
                <input type="hidden" name="productId" value={prod.id} />
                <button type="submit" className="btn-order">سفارش آنی و استقرار</button>
              </form>
            </article>
          ))}
        </section>
      </Suspense>
    </main>
  );
}

'use server';

import { revalidateTag, revalidatePath } from 'next/cache';

export async function orderServerAction(formData: FormData) {
  const productId = formData.get('productId');
  if (!productId) return { error: 'شناسه محصول الزامی است.' };

  // Secure server-side business logic
  await db.orders.create({
    data: {
      productId: String(productId),
      timestamp: new Date()
    }
  });

  // Revalidate multi-layer cache instantly
  revalidateTag('products');
  revalidatePath('/dashboard');

  return { success: true };
}

شماره خط سینتکس و عبارت کد نقش فنی در سیستم شرح عملکرد و جزئیات مهندسی
خط ۶ import { Suspense } from 'react'; واردسازی قابلیت استریمینگ ناهمگام ری‌اکت تگ Suspense در Next.js 15 اجازه می‌دهد بخش‌های سریع صفحه بلافاصله لود شوند و بخش‌های سنگین بدون بلاک کردن کل صفحه، هم‌زمان به مرورگر استریم گردند.
خط ۱۰ async function getEnterpriseProducts() { ... } دریافت مستقیم داده در سرور بدون API کلاینت در React Server Components (RSC) کامپوننت خود یک تابع async است؛ بنابراین بدون نیاز به useState و useEffect مستقیماً دیتا را از دیتابیس می‌خواند.
خط ۱۳ next: { tags: ['products'], revalidate: 3600 } کش‌گذاری تگ‌دار و هوشمند ISR در سطح Edge داده‌ها به مدت ۳۶۰۰ ثانیه در کش لبه CDN می‌مانند، اما هر زمان ادمین تغییری دهد می‌توان با revalidateTag('products') فوراً کش را بدون انتظار باطل کرد.
خط ۱۸ export default async function HomePage() { ... } کامپوننت غیرهمگام سرور با صفر بایت کد JS ارسالی کدهای این کامپوننت کاملاً در سرور اجرا و به HTML خالص تبدیل می‌شوند؛ یعنی هیچ فایل جاوا اسکریپتی برای این منطق به موبایل کاربر ارسال نمی‌شود.
خط ۲۸ <Suspense fallback={<ServerMetricsSkeleton />}> پوسته پیش‌رندر جزئی (PPR Fallback) تا زمان دریافت داده‌های محصولات از سرور، اسکلتون متحرک و زیبایی به کاربر نشان داده می‌شود تا امتیاز Largest Contentful Paint (LCP) سبز باشد.
خط ۳۵ <form action={orderServerAction}> اتصال مستقیم فرم HTML به تابع بک‌اند انقلاب سرور اکشن‌ها: بدون نوشتن حتی یک خط کد fetch یا تعریف اندپوینت API، ارسال فرم مستقیماً متد سرور orderServerAction را با متد POST صدا می‌زند.
خط ۴۹ 'use server'; مرز امنیتی سرور و کلاینت دستور جادویی کامپایلر Next.js: به توربوپک اعلام می‌کند تمام توابع داخل این فایل فقط و فقط روی سرور نود یا Edge اجرا شوند و در باندل عمومی فرانت‌اند نروند.
خط ۶۴ revalidateTag('products'); باطل‌سازی هدفمند کش در CDN بلافاصله پس از ثبت سفارش، کش دیتای محصولات در سراسر سرورهای CDN لغو می‌شود تا موجودی انبار به ثانیه‌ای به‌روزرسانی شود.
